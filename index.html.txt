<script>
    // --- 🎵 SONIDOS PRECARGADOS ---
    const audioClick = new Audio("click.mp3");
    const audioSuccess = new Audio("exito.mp3");
    audioClick.volume = 0.35;
    audioSuccess.volume = 0.55;

    function sonidoClick() {
        audioClick.currentTime = 0;
        audioClick.play().catch(() => {});
    }

    function sonidoExito() {
        audioSuccess.currentTime = 0;
        audioSuccess.play().catch(() => {});
    }

    // --- CONSTANTES DE PRECIO ---
    const PRECIO_CAMBIO_EXTRA = 3;
    const PRECIO_PACK_PROMO = 20;
    const PRECIO_MAQ_SIMPLE = 0.50; 
    const PRECIO_MAQ_DECORADA = 0.75; 
    const PRECIO_MAQ_VISUAL = 1.00; 

    window.onload = function() {
        toggleMetodoPago(); 
        cargarTasa();       
    };

    function toggleCheckbox(id) {
        const chk = document.getElementById(id);
        chk.checked = !chk.checked;
        sonidoClick();
        calcular();
    }

    function cargarTasa() {
        // INTENTO 1: DolarApi oficial
        fetch('https://ve.dolarapi.com/v1/euros/oficial')
            .then(res => res.json())
            .then(data => {
                actualizarTasa(data.promedio || data.price);
            })
            .catch(() => {
                // INTENTO 2: PyDolarVenezuela
                fetch('https://pydolarve.org/api/v1/dollar?page=bcv')
                    .then(res => res.json())
                    .then(data => {
                        const precio = data.monitors && data.monitors.eur ? data.monitors.eur.price : null;
                        if (precio) actualizarTasa(precio);
                        else throw new Error("No eur price");
                    })
                    .catch(() => {
                        document.getElementById('status-dot').className = 'dot error';
                        document.getElementById('status-text').innerText = "Ingresa la tasa BCV manualmente.";
                        document.getElementById('tasa').placeholder = "0.00";
                    });
            });
    }

    function actualizarTasa(precio) {
        document.getElementById('tasa').value = precio;
        document.getElementById('status-dot').className = 'dot active';
        document.getElementById('status-text').innerText = "Tasa BCV (Euro) en Vivo";
        document.getElementById('status-text').style.color = "#00ff88";
        calcular(); 
    }

    function toggleMetodoPago() {
        const metodo = document.getElementById("metodo-pago").value;
        const bloqueTasa = document.getElementById("bloque-tasa");
        const bloqueBs = document.getElementById("bloque-bs");
        const infoInt = document.getElementById("info-internacional");

        if (metodo === "internacional") {
            bloqueTasa.style.display = "none";
            bloqueBs.style.display = "none";
            infoInt.classList.remove("hidden");
        } else {
            bloqueTasa.style.display = "block";
            bloqueBs.style.display = "flex";
            infoInt.classList.add("hidden");
        }
        calcular();
    }

    function mostrarOpciones() {
        const servicio = document.getElementById("servicio").value;
        const opcPortada = document.getElementById("opciones-portada");
        const opcMaq = document.getElementById("opciones-maquetacion");
        
        opcPortada.classList.add("hidden");
        opcMaq.classList.add("hidden");

        if (servicio === "portada") {
            opcPortada.classList.remove("hidden");
        } else if (servicio === "maquetacion") {
            opcMaq.classList.remove("hidden");
        } else if (servicio === "combo") {
            opcPortada.classList.remove("hidden");
            opcMaq.classList.remove("hidden");
        }
        calcular();
    }

    function toggleMedida() {
        const tipo = document.getElementById("tipo-maquetacion").value;
        if (tipo === "agenda") {
            document.getElementById("input-palabras").classList.add("hidden");
            document.getElementById("input-paginas").classList.remove("hidden");
        } else {
            document.getElementById("input-palabras").classList.remove("hidden");
            document.getElementById("input-paginas").classList.add("hidden");
        }
        calcular();
    }

    function calcular() {
        const servicio = document.getElementById("servicio").value;
        let total = 0;
        let itemsBreakdown = [];

        // 1. Cálculo de Portada
        if (servicio === "portada" || servicio === "combo") {
            const selectPort = document.getElementById("tipo-portada");
            const base = parseFloat(selectPort.value) || 0;
            const textPort = selectPort.selectedIndex > 0 ? selectPort.options[selectPort.selectedIndex].text.split(' - ')[0] : 'Portada';
            
            if (base > 0) {
                total += base;
                itemsBreakdown.push({ label: textPort, cost: base });
            }

            // Pack Promocional
            const packPromo = document.getElementById("check-pack-promo").checked;
            if (packPromo) {
                total += PRECIO_PACK_PROMO;
                itemsBreakdown.push({ label: "Pack Promo (+Mockup 3D + Banners)", cost: PRECIO_PACK_PROMO });
            }

            // Extras
            const extras = parseInt(document.getElementById("cambios-extra").value) || 0;
            if (extras > 0) {
                const costoExtras = extras * PRECIO_CAMBIO_EXTRA;
                total += costoExtras;
                itemsBreakdown.push({ label: `${extras} cambio(s) adicional(es)`, cost: costoExtras });
            }
        } 

        // 2. Cálculo de Maquetación
        if (servicio === "maquetacion" || servicio === "combo") {
            const tipo = document.getElementById("tipo-maquetacion").value;
            let subtotalMaq = 0;
            let descMaq = "";
            
            if (tipo === "agenda") {
                const paginas = parseInt(document.getElementById("cantidad-paginas").value) || 0;
                subtotalMaq = paginas * PRECIO_MAQ_VISUAL;
                descMaq = `Maquetación Agenda (${paginas} págs)`;
            } else {
                const palabras = parseInt(document.getElementById("cantidad-palabras").value) || 0;
                const tasaPalabras = (tipo === "simple") ? PRECIO_MAQ_SIMPLE : PRECIO_MAQ_DECORADA;
                subtotalMaq = (palabras / 1000) * tasaPalabras;
                const tipoNombre = (tipo === "simple") ? "Fluida" : "Decorada";
                descMaq = `Maquetación ${tipoNombre} (${palabras.toLocaleString()} palabras)`;
            }

            if (subtotalMaq > 0 && subtotalMaq < 10) {
                subtotalMaq = 10;
                descMaq += " [Mínimo aplicado]";
            }

            if (subtotalMaq > 0) {
                total += subtotalMaq;
                itemsBreakdown.push({ label: descMaq, cost: subtotalMaq });
            }
        }

        // 3. Descuento Combo 10%
        if (servicio === "combo" && total > 0) {
            const descuento = total * 0.10;
            total -= descuento;
            itemsBreakdown.push({ label: "Descuento Combo Especial (10%)", cost: -descuento, isDiscount: true });
        }

        // Render Desglose
        const bList = document.getElementById("breakdown-list");
        bList.innerHTML = "";
        if (itemsBreakdown.length > 0) {
            bList.classList.remove("hidden");
            itemsBreakdown.forEach(item => {
                const li = document.createElement("li");
                li.className = "breakdown-item" + (item.isDiscount ? " highlight" : "");
                li.innerHTML = `<span>${item.label}</span><span>${item.cost > 0 ? '+' : ''}${item.cost.toFixed(2)} USD</span>`;
                bList.appendChild(li);
            });
        } else {
            bList.classList.add("hidden");
        }

        // Render Total USD/EUR
        document.getElementById("total-ref").innerText = total.toFixed(2);

        // Render Total Bolívares
        const metodo = document.getElementById("metodo-pago").value;
        if (metodo === "nacional") {
            const tasa = parseFloat(document.getElementById("tasa").value) || 0;
            const totalBs = total * tasa;
            const bsFormateado = new Intl.NumberFormat('es-VE', { style: 'currency', currency: 'VES' }).format(totalBs);
            document.getElementById("total-bs").innerText = bsFormateado;
        }
    }

    function enviarWhatsapp() {
        const servicio = document.getElementById("servicio").value;
        const total = document.getElementById("total-ref").innerText;
        const totalBs = document.getElementById("total-bs").innerText;
        const tasa = document.getElementById("tasa").value;
        const metodo = document.getElementById("metodo-pago").value;
        const telefono = "584125796738"; 

        if (servicio === "none") {
            alert("Por favor selecciona un servicio para cotizar.");
            return;
        }

        let resumenServicio = "";
        if (servicio === "portada") {
            const selPort = document.getElementById("tipo-portada");
